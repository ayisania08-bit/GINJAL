<script>
const rules = [
{
 nama:"Gagal Ginjal Kronis",
 gejala:["G11","G14","G22"],
 solusi:"• Diet rendah garam\n• Kontrol rutin ke dokter\n• Terapi jangka panjang"
},
{
 nama:"Batu Ginjal",
 gejala:["G1","G2","G14"],
 solusi:"• Minum air putih minimal 2 liter\n• Hindari makanan tinggi oksalat"
},
{
 nama:"Infeksi Ginjal",
 gejala:["G2","G3","G14"],
 solusi:"• Konsumsi antibiotik sesuai anjuran\n• Jaga kebersihan saluran kemih"
}
];

function diagnosa(){
    let input=[];

    document.querySelectorAll('input[type=radio]:checked').forEach(r=>{
        if(r.value==="ya"){
            input.push(r.name);
        }
    });

    let hasil="";
    rules.forEach(r=>{
        if(r.gejala.every(g=>input.includes(g))){
            hasil+=`
            <div class="penyakit">
                <b>${r.nama}</b>
                <div class="solusi">${r.solusi.replace(/\n/g,"<br>")}</div>
            </div>`;
        }
    });

    document.getElementById("hasil").innerHTML =
        hasil || "Tidak ditemukan penyakit yang sesuai berdasarkan jawaban Anda.";
}
</script>
